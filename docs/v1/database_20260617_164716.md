Berikut adalah contoh SQLAlchemy models untuk semua tabel MVP:
```python
from sqlalchemy import Column, Integer, String, UUID, JSON, DateTime, Date, Boolean, Float
from sqlalchemy.dialects.postgresql import UUID as PG_UUID
from sqlalchemy.orm import relationship
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.sql import func

Base = declarative_base()

class Tenant(Base):
    __tablename__ = 'tenants'
    id = Column(PG_UUID(as_uuid=True), primary_key=True, default=func.uuid_generate_v4())
    name = Column(String, nullable=False)
    slug = Column(String, nullable=False, unique=True)
    plan_id = Column(PG_UUID(as_uuid=True), nullable=False)
    settings = Column(JSON, nullable=False, default={})
    created_at = Column(DateTime, nullable=False, server_default=func.now())
    deleted_at = Column(DateTime, nullable=True)

    users = relationship('User', backref='tenant')
    ai_agents = relationship('AiAgent', backref='tenant')
    conversations = relationship('Conversation', backref='tenant')
    knowledge_bases = relationship('KnowledgeBase', backref='tenant')
    customers = relationship('Customer', backref='tenant')
    subscriptions = relationship('Subscription', backref='tenant')

class User(Base):
    __tablename__ = 'users'
    id = Column(PG_UUID(as_uuid=True), primary_key=True, default=func.uuid_generate_v4())
    tenant_id = Column(PG_UUID(as_uuid=True), nullable=False)
    email = Column(String, nullable=False, unique=True)
    name = Column(String, nullable=False)
    avatar_url = Column(String, nullable=True)
    provider = Column(String, nullable=False)
    role = Column(String, nullable=False)
    is_verified = Column(Boolean, nullable=False, default=False)
    created_at = Column(DateTime, nullable=False, server_default=func.now())
    deleted_at = Column(DateTime, nullable=True)

    otp_codes = relationship('OtpCode', backref='user')
    conversations = relationship('Conversation', backref='user')

class OtpCode(Base):
    __tablename__ = 'otp_codes'
    id = Column(PG_UUID(as_uuid=True), primary_key=True, default=func.uuid_generate_v4())
    user_id = Column(PG_UUID(as_uuid=True), nullable=False)
    channel = Column(String, nullable=False)
    code = Column(String, nullable=False)
    expires_at = Column(DateTime, nullable=False)
    used_at = Column(DateTime, nullable=True)
    created_at = Column(DateTime, nullable=False, server_default=func.now())

class AiAgent(Base):
    __tablename__ = 'ai_agents'
    id = Column(PG_UUID(as_uuid=True), primary_key=True, default=func.uuid_generate_v4())
    tenant_id = Column(PG_UUID(as_uuid=True), nullable=False)
    name = Column(String, nullable=False)
    type = Column(String, nullable=False)
    system_prompt = Column(String, nullable=False)
    config = Column(JSON, nullable=False, default={})
    is_active = Column(Boolean, nullable=False, default=True)
    created_at = Column(DateTime, nullable=False, server_default=func.now())
    deleted_at = Column(DateTime, nullable=True)

    conversations = relationship('Conversation', backref='ai_agent')
    knowledge_bases = relationship('KnowledgeBase', backref='ai_agent')

class Conversation(Base):
    __tablename__ = 'conversations'
    id = Column(PG_UUID(as_uuid=True), primary_key=True, default=func.uuid_generate_v4())
    tenant_id = Column(PG_UUID(as_uuid=True), nullable=False)
    ai_agent_id = Column(PG_UUID(as_uuid=True), nullable=False)
    customer_id = Column(PG_UUID(as_uuid=True), nullable=False)
    channel = Column(String, nullable=False)
    status = Column(String, nullable=False)
    created_at = Column(DateTime, nullable=False, server_default=func.now())

    messages = relationship('Message', backref='conversation')

class Message(Base):
    __tablename__ = 'messages'
    id = Column(PG_UUID(as_uuid=True), primary_key=True, default=func.uuid_generate_v4())
    conversation_id = Column(PG_UUID(as_uuid=True), nullable=False)
    role = Column(String, nullable=False)
    content = Column(String, nullable=False)
    metadata = Column(JSON, nullable=False, default={})
    created_at = Column(DateTime, nullable=False, server_default=func.now())

class KnowledgeBase(Base):
    __tablename__ = 'knowledge_bases'
    id = Column(PG_UUID(as_uuid=True), primary_key=True, default=func.uuid_generate_v4())
    tenant_id = Column(PG_UUID(as_uuid=True), nullable=False)
    ai_agent_id = Column(PG_UUID(as_uuid=True), nullable=False)
    name = Column(String, nullable=False)
    created_at = Column(DateTime, nullable=False, server_default=func.now())

    kb_documents = relationship('KbDocument', backref='knowledge_base')

class KbDocument(Base):
    __tablename__ = 'kb_documents'
    id = Column(PG_UUID(as_uuid=True), primary_key=True, default=func.uuid_generate_v4())
    knowledge_base_id = Column(PG_UUID(as_uuid=True), nullable=False)
    title = Column(String, nullable=False)
    content = Column(String, nullable=False)
    embedding = Column(Float(precision=32, scale=10), nullable=False)
    created_at = Column(DateTime, nullable=False, server_default=func.now())

class Customer(Base):
    __tablename__ = 'customers'
    id = Column(PG_UUID(as_uuid=True), primary_key=True, default=func.uuid_generate_v4())
    tenant_id = Column(PG_UUID(as_uuid=True), nullable=False)
    name = Column(String, nullable=False)
    phone = Column(String, nullable=False)
    email = Column(String, nullable=False)
    channel = Column(String, nullable=False)
    last_contact_at = Column(DateTime, nullable=True)
    created_at = Column(DateTime, nullable=False, server_default=func.now())

    conversations = relationship('Conversation', backref='customer')

class Plan(Base):
    __tablename__ = 'plans'
    id = Column(PG_UUID(as_uuid=True), primary_key=True, default=func.uuid_generate_v4())
    name = Column(String, nullable=False)
    slug = Column(String, nullable=False, unique=True)
    price_monthly = Column(Integer, nullable=False)
    limits = Column(JSON, nullable=False, default={})
    is_active = Column(Boolean, nullable=False, default=True)
    created_at = Column(DateTime, nullable=False, server_default=func.now())

    subscriptions = relationship('Subscription', backref='plan')

class Subscription(Base):
    __tablename__ = 'subscriptions'
    id = Column(PG_UUID(as_uuid=True), primary_key=True, default=func.uuid_generate_v4())
    tenant_id = Column(PG_UUID(as_uuid=True), nullable=False)
    plan_id = Column(PG_UUID(as_uuid=True), nullable=False)
    status = Column(String, nullable=False)
    current_period_start = Column(DateTime, nullable=False)
    current_period_end = Column(DateTime, nullable=False)
    created_at = Column(DateTime, nullable=False, server_default=func.now())

    invoices = relationship('Invoice', backref='subscription')

class Invoice(Base):
    __tablename__ = 'invoices'
    id = Column(PG_UUID(as_uuid=True), primary_key=True, default=func.uuid_generate_v4())
    tenant_id = Column(PG_UUID(as_uuid=True), nullable=False)
    subscription_id = Column(PG_UUID(as_uuid=True), nullable=False)
    amount = Column(Integer, nullable=False)
    status = Column(String, nullable=False)
    paid_at = Column(DateTime, nullable=True)
    payment_method = Column(String, nullable=False)
    created_at = Column(DateTime, nullable=False, server_default=func.now())

class UsageLog(Base):
    __tablename__ = 'usage_logs'
    id = Column(PG_UUID(as_uuid=True), primary_key=True, default=func.uuid_generate_v4())
    tenant_id = Column(PG_UUID(as_uuid=True), nullable=False)
    metric = Column(String, nullable=False)
    count = Column(Integer, nullable=False)
    period_start = Column(Date, nullable=False)
    created_at = Column(DateTime, nullable=False, server_default=func.now())
```
Perhatikan bahwa saya menggunakan `PG_UUID` dari `sqlalchemy.dialects.postgresql` untuk membuat kolom UUID, dan `JSON` untuk membuat kolom JSON. Saya juga menggunakan `Float(precision=32, scale=10)` untuk membuat kolom floating-point dengan presisi 32 dan skala 10.
