```python:backend/app/modules/auth/models.py
from sqlalchemy.orm import relationship

Base = declarative_base()

class User(Base):
    __tablename__ = 'users'

    id = Column(Integer, primary_key=True)
    email = Column(String, unique=True)
    password = Column(String)
    name = Column(String)
    is_active = Column(Boolean, default=True)
    is_oauth = Column(Boolean, default=False)
    oauth_provider = Column(String)

    def __repr__(self):
        return f"User(id={self.id}, email={self.email}, name={self.name})"
```

```python:backend/app/modules/auth/router.py
from fastapi import APIRouter, Depends, HTTPException
from typing import Optional
from .. import schemas, service
from .. import utils
from sqlalchemy.orm import Session
from ..database import get_db

router = APIRouter(prefix="/auth", tags=["auth"])

@router.post("/register")
async def register(user: schemas.UserRegister, db: Session = Depends(get_db)):
    auth_service = service.AuthService(db)
    user_obj = auth_service.register(user)
    if not user_obj:
        raise HTTPException(status_code=400, detail="Gagal registrasi")
    return {"success": True, "data": user_obj, "message": "Berhasil registrasi"}

@router.post("/login")
async def login(user: schemas.UserLogin, db: Session = Depends(get_db)):
    auth_service = service.AuthService(db)
    user_obj = auth_service.login(user)
    if not user_obj:
        raise HTTPException(status_code=401, detail="Invalid email or password")
    return {"success": True, "data": user_obj, "message": "Berhasil login"}

@router.post("/oauth/login")
async def oauth_login(user: schemas.UserOAuth, db: Session = Depends(get_db)):
    auth_service = service.AuthService(db)
    user_obj = auth_service.oauth_login(user)
    if not user_obj:
        raise HTTPException(status_code=401, detail="Invalid OAuth credentials")
    return {"success": True, "data": user_obj, "message": "Berhasil login dengan OAuth"}

@router.post("/otp/request")
async def request_otp(request: schemas.OTPRequest, db: Session = Depends(get_db)):
    auth_service = service.AuthService(db)
    auth_service.request_otp(request)
    return {"success": True, "message": "Kode OTP telah dikirim"}

@router.post("/otp/verify")
async def verify_otp(request: schemas.OTPVerify, db: Session = Depends(get_db)):
    auth_service = service.AuthService(db)
    auth_service.verify_otp(request)
```

```python:backend/app/middleware/auth.py
from fastapi import Request, HTTPException
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from .. import utils
from ..database import get_db
from sqlalchemy.orm import Session

auth_scheme = HTTPBearer()

async def get_current_user(request: Request, token: HTTPAuthorizationCredentials = Depends(auth_scheme), db: Session = Depends(get_db)):
    credentials_exception = HTTPException(
        status_code=401,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        payload = utils.decode_token(token.credentials)
        user = db.query(User).filter(User.email == payload.email).first()
        if user is None:
            raise credentials_exception
        return user
    except Exception as e:
        raise credentials_exception
```

```python:backend/app/services/otp_service.py
from typing import Optional
from .. import utils
import random

class OTPServices:
    def __init__(self):
        self.otp_codes = {}

    def generate_otp(self, email: str):
        otp_code = random.randint(100000, 999999)
        self.otp_codes[email] = otp_code
        return otp_code

    def verify_otp(self, email: str, otp_code: int):
        if email not in self.otp_codes:
            return False
        return self.otp_codes[email] == otp_code
```
