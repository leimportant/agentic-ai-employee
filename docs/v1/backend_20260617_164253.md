Berikut adalah contoh implementasi module ai_agents dengan CRUD agent dan conversation endpoint menggunakan FastAPI:

**modules/ai_agents/models.py**
```python
from sqlalchemy import Column, Integer, String, ForeignKey
from sqlalchemy.orm import relationship
from backend.database import Base

class AiAgent(Base):
    __tablename__ = "ai_agents"

    id = Column(Integer, primary_key=True)
    name = Column(String, nullable=False)
    description = Column(String, nullable=True)
    tenant_id = Column(Integer, ForeignKey("tenants.id"), nullable=False)

    conversations = relationship("AiAgentConversation", back_populates="agent")

class AiAgentConversation(Base):
    __tablename__ = "ai_agent_conversations"

    id = Column(Integer, primary_key=True)
    agent_id = Column(Integer, ForeignKey("ai_agents.id"), nullable=False)
    conversation_id = Column(String, nullable=False)
    agent = relationship("AiAgent", back_populates="conversations")
```

**modules/ai_agents/schemas.py**
```python
from pydantic import BaseModel

class AiAgentCreate(BaseModel):
    name: str
    description: str | None

class AiAgentUpdate(BaseModel):
    name: str | None
    description: str | None

class AiAgentResponse(BaseModel):
    id: int
    name: str
    description: str | None

class AiAgentConversationCreate(BaseModel):
    conversation_id: str

class AiAgentConversationResponse(BaseModel):
    id: int
    conversation_id: str
```

**modules/ai_agents/service.py**
```python
from backend.database import SessionLocal
from modules.ai_agents.models import AiAgent, AiAgentConversation

class AiAgentService:
    def get_all_agents(self, db: SessionLocal, tenant_id: int):
        return db.query(AiAgent).filter(AiAgent.tenant_id == tenant_id).all()

    def get_agent(self, db: SessionLocal, agent_id: int, tenant_id: int):
        return db.query(AiAgent).filter(AiAgent.id == agent_id, AiAgent.tenant_id == tenant_id).first()

    def create_agent(self, db: SessionLocal, agent_create: AiAgentCreate, tenant_id: int):
        agent = AiAgent(name=agent_create.name, description=agent_create.description, tenant_id=tenant_id)
        db.add(agent)
        db.commit()
        db.refresh(agent)
        return agent

    def update_agent(self, db: SessionLocal, agent_id: int, agent_update: AiAgentUpdate, tenant_id: int):
        agent = db.query(AiAgent).filter(AiAgent.id == agent_id, AiAgent.tenant_id == tenant_id).first()
        if agent:
            agent.name = agent_update.name or agent.name
            agent.description = agent_update.description or agent.description
            db.commit()
            db.refresh(agent)
            return agent
        return None

    def delete_agent(self, db: SessionLocal, agent_id: int, tenant_id: int):
        agent = db.query(AiAgent).filter(AiAgent.id == agent_id, AiAgent.tenant_id == tenant_id).first()
        if agent:
            db.delete(agent)
            db.commit()
            return True
        return False

    def get_all_conversations(self, db: SessionLocal, agent_id: int, tenant_id: int):
        return db.query(AiAgentConversation).filter(AiAgentConversation.agent_id == agent_id).all()

    def create_conversation(self, db: SessionLocal, conversation_create: AiAgentConversationCreate, agent_id: int, tenant_id: int):
        conversation = AiAgentConversation(conversation_id=conversation_create.conversation_id, agent_id=agent_id)
        db.add(conversation)
        db.commit()
        db.refresh(conversation)
        return conversation
```

**modules/ai_agents/router.py**
```python
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel
from backend.database import SessionLocal
from modules.ai_agents.service import AiAgentService
from modules.ai_agents.schemas import AiAgentCreate, AiAgentUpdate, AiAgentResponse, AiAgentConversationCreate, AiAgentConversationResponse

router = APIRouter(prefix="/ai-agents", tags=["Ai Agents"])

@router.get("/")
async def get_all_agents(tenant_id: int, db: SessionLocal = Depends()):
    ai_agent_service = AiAgentService()
    agents = ai_agent_service.get_all_agents(db, tenant_id)
    return {"success": True, "data": [AiAgentResponse(id=agent.id, name=agent.name, description=agent.description) for agent in agents]}

@router.get("/{agent_id}")
async def get_agent(agent_id: int, tenant_id: int, db: SessionLocal = Depends()):
    ai_agent_service = AiAgentService()
    agent = ai_agent_service.get_agent(db, agent_id, tenant_id)
    if agent:
        return {"success": True, "data": AiAgentResponse(id=agent.id, name=agent.name, description=agent.description)}
    raise HTTPException(status_code=404, detail="Agent not found")

@router.post("/")
async def create_agent(agent_create: AiAgentCreate, tenant_id: int, db: SessionLocal = Depends()):
    ai_agent_service = AiAgentService()
    agent = ai_agent_service.create_agent(db, agent_create, tenant_id)
    return {"success": True, "data": AiAgentResponse(id=agent.id, name=agent.name, description=agent.description)}

@router.put("/{agent_id}")
async def update_agent(agent_id: int, agent_update: AiAgentUpdate, tenant_id: int, db: SessionLocal = Depends()):
    ai_agent_service = AiAgentService()
    agent = ai_agent_service.update_agent(db, agent_id, agent_update, tenant_id)
    if agent:
        return {"success": True, "data": AiAgentResponse(id=agent.id, name=agent.name, description=agent.description)}
    raise HTTPException(status_code=404, detail="Agent not found")

@router.delete("/{agent_id}")
async def delete_agent(agent_id: int, tenant_id: int, db: SessionLocal = Depends()):
    ai_agent_service = AiAgentService()
    if ai_agent_service.delete_agent(db, agent_id, tenant_id):
        return {"success": True}
    raise HTTPException(status_code=404, detail="Agent not found")

@router.get("/{agent_id}/conversations")
async def get_all_conversations(agent_id: int, tenant_id: int, db: SessionLocal = Depends()):
    ai_agent_service = AiAgentService()
    conversations = ai_agent_service.get_all_conversations(db, agent_id, tenant_id)
    return {"success": True, "data": [AiAgentConversationResponse(id=conversation.id, conversation_id=conversation.conversation_id) for conversation in conversations]}

@router.post("/{agent_id}/conversations")
async def create_conversation(conversation_create: AiAgentConversationCreate, agent_id: int, tenant_id: int, db: SessionLocal = Depends()):
    ai_agent_service = AiAgentService()
    conversation = ai_agent_service.create_conversation(db, conversation_create, agent_id, tenant_id)
    return {"success": True, "data": AiAgentConversationResponse(id=conversation.id, conversation_id=conversation.conversation_id)}
```
