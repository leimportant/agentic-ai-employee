```frontend/app/(auth)/register/page.tsx
import { useClient } from 'next/navigation';
import { useState } from 'react';
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Chrome } from "lucide-react";
import Link from 'next/link';
import axios from 'axios';

use client;

export default function RegisterPage() {
  const [email, setEmail] = useState('');
  const [otp, setOtp] = useState('');
  const [isOtpSent, setIsOtpSent] = useState(false);

  const handleGoogleOAuth = async () => {
    const response = await axios.get('/api/v1/auth/google');
    window.location.href = response.data.url;
  };

  const handleOtpSend = async () => {
    try {
      const response = await axios.post('/api/v1/auth/otp', { email });
      setIsOtpSent(true);
    } catch (error) {
      console.error(error);
    }
  };

  const handleOtpVerify = async () => {
    try {
      const response = await axios.post('/api/v1/auth/otp/verify', { email, otp });
      if (response.data.success) {
        // Redirect to login page
        window.location.href = '/auth/login';
      }
    } catch (error) {
      console.error(error);
    }
  };

  return (
    <div className="flex justify-center items-center h-screen">
      <div className="bg-slate-800 p-6 rounded-md w-96">
        <h2 className="text-slate-50 text-2xl mb-4">Daftar</h2>
        <Button onClick={handleGoogleOAuth} className="flex items-center gap-2 mb-4">
          <Chrome size={20} /> Masuk dengan Google
        </Button>
        <div className="mb-4">
          <Input
            type="email"
            value={email}
            onChange={(e) => setEmail(e.target.value)}
            placeholder="Email"
          />
        </div>
        {isOtpSent ? (
          <div className="mb-4">
            <Input
              type="number"
              value={otp}
              onChange={(e) => setOtp(e.target.value)}
              placeholder="Kode OTP"
            />
            <Button onClick={handleOtpVerify} className="mt-2">Verifikasi</Button>
          </div>
        ) : (
          <Button onClick={handleOtpSend}>Kirim Kode OTP</Button>
        )}
        <p className="text-slate-400 mt-4">
          Sudah memiliki akun? <Link href="/auth/login" className="text-indigo-500">Masuk</Link>
        </p>
      </div>
    </div>
  );
}
```
