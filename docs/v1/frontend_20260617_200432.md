import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { useState } from "react";
import { useZustandStore } from "@/store";
import { useForm } from "react-hook-form";
import { z } from "zod";
import { zodResolver } from "@hookform/resolvers/zod";
import { GoogleOAuthButton } from "@/components/ui/google-oauth-button";
import { OTPVerificationForm } from "@/components/ui/otp-verification-form";
import Link from "next/link";

const LoginPage = () => {
  const { login } = useZustandStore();
  const [isGoogleAuth, setIsGoogleAuth] = useState(false);
  const [otpVerification, setOtpVerification] = useState(false);
  const { register, handleSubmit, formState: { errors } } = useForm({
    resolver: zodResolver(z.object({
      email: z.string().email(),
      password: z.string().min(8),
    })),
  });

  const handleGoogleAuth = async () => {
    setIsGoogleAuth(true);
  };

  const handleOtpVerification = async () => {
    setOtpVerification(true);
  };

  const onSubmit = async (data) => {
    await login(data);
  };

  return (
    <div className="flex justify-center items-center h-screen">
      <div className="card w-96 bg-slate-800 shadow-xl">
        <div className="card-body">
          <h2 className="card-title text-slate-50">Masuk</h2>
          {isGoogleAuth ? (
            <GoogleOAuthButton onAuth={handleOtpVerification} />
          ) : (
            <form onSubmit={handleSubmit(onSubmit)}>
              <Input
                type="email"
                placeholder="Email"
                {...register("email")}
                error={errors.email?.message}
              />
              <Input
                type="password"
                placeholder="Password"
                {...register("password")}
                error={errors.password?.message}
              />
              <Button type="submit" className="btn btn-block">Masuk</Button>
            </form>
          )}
          {otpVerification && (
            <OTPVerificationForm />
          )}
          <p className="text-slate-400">
            Belum memiliki akun? <Link href="/register">Daftar</Link>
          </p>
          <Button className="btn btn-block" onClick={handleGoogleAuth}>
            Masuk dengan Google
          </Button>
        </div>
      </div>
    </div>
  );
};

export default LoginPage;
