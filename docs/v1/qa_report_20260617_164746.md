**Test Cases untuk Billing Module**
=====================================

Berikut adalah beberapa test cases untuk billing module menggunakan pytest:

### Test Case 1: Plan Limits Enforcement

* **Test Case:** `test_plan_limits_enforcement`
* **Deskripsi:** Menguji apakah plan limits diterapkan dengan benar
* **Kode:**
```python
import pytest
from fastapi import FastAPI
from app.models import Plan
from app.billing import check_plan_limits

@pytest.fixture
def app():
    return FastAPI()

def test_plan_limits_enforcement(app):
    # Buat plan dengan limit 10
    plan = Plan(name="Plan 1", limit=10)
    
    # Coba membuat 11 item
    for i in range(11):
        item = {"name": f"Item {i}"}
        check_plan_limits(plan, item)
    
    # Harusnya gagal di item ke-11
    with pytest.raises(Exception):
        check_plan_limits(plan, {"name": "Item 11"})
```

### Test Case 2: Upgrade/Downgrade Plan

* **Test Case:** `test_upgrade_downgrade_plan`
* **Deskripsi:** Menguji apakah upgrade dan downgrade plan berfungsi dengan benar
* **Kode:**
```python
import pytest
from fastapi import FastAPI
from app.models import Plan, User
from app.billing import upgrade_plan, downgrade_plan

@pytest.fixture
def app():
    return FastAPI()

def test_upgrade_downgrade_plan(app):
    # Buat user dengan plan awal
    user = User(email="user@example.com", plan=Plan(name="Plan 1", limit=10))
    
    # Upgrade plan ke plan yang lebih besar
    new_plan = Plan(name="Plan 2", limit=20)
    upgrade_plan(user, new_plan)
    assert user.plan == new_plan
    
    # Downgrade plan ke plan yang lebih kecil
    old_plan = Plan(name="Plan 1", limit=10)
    downgrade_plan(user, old_plan)
    assert user.plan == old_plan
```

### Test Case 3: Webhook Payment

* **Test Case:** `test_webhook_payment`
* **Deskripsi:** Menguji apakah webhook payment berfungsi dengan benar
* **Kode:**
```python
import pytest
from fastapi import FastAPI
from app.models import Payment
from app.billing import process_webhook_payment

@pytest.fixture
def app():
    return FastAPI()

def test_webhook_payment(app):
    # Buat payment dengan status pending
    payment = Payment(amount=10, status="pending")
    
    # Simulasi webhook payment
    webhook_data = {"event": "payment.success", "payment_id": payment.id}
    process_webhook_payment(webhook_data)
    assert payment.status == "success"
```
